## 61. Move All Zeroes To End (LeetCode 283)

```java
import java.util.Arrays;

public class MoveZeroes {

    public static void main(String[] args) {

        int[] nums = {0, 1, 0, 3, 12};

        moveZeroes(nums);

        System.out.println(
                Arrays.toString(nums)
        );
    }

    static void moveZeroes(int[] nums) {

        int j = 0;

        // Move all non-zero elements to front
        for (int i = 0; i < nums.length; i++) {

            if (nums[i] != 0) {

                int temp = nums[i];
                nums[i] = nums[j];
                nums[j] = temp;

                j++;
            }
        }
    }
}
```

## 62. Move All Zeroes To Beginning

```java
import java.util.Arrays;

public class MoveZeroesToBeginning {

    public static void main(String[] args) {

        int[] nums = {1, 0, 3, 0, 5, 0, 7};

        int j = nums.length - 1;

        // Move non-zero values to end
        for (int i = nums.length - 1; i >= 0; i--) {

            if (nums[i] != 0) {

                int temp = nums[i];
                nums[i] = nums[j];
                nums[j] = temp;

                j--;
            }
        }

        System.out.println(
                Arrays.toString(nums)
        );
    }
}
```

## 63. Count Characters That Can Form a Word (LeetCode 1160)

```java
import java.util.HashMap;
import java.util.Map;

public class CountCharacters {

    public static void main(String[] args) {

        String[] words =
                {"cat", "bt", "hat", "tree"};

        String chars = "atach";

        System.out.println(
                countCharacters(words, chars)
        );
    }

    static int countCharacters(
            String[] words,
            String chars) {

        int totalLength = 0;

        // Character frequency of source string
        Map<Character, Integer> source =
                new HashMap<>();

        for (char ch : chars.toCharArray()) {

            source.put(
                    ch,
                    source.getOrDefault(ch, 0) + 1
            );
        }

        for (String word : words) {

            if (canForm(word, source)) {

                totalLength += word.length();
            }
        }

        return totalLength;
    }

    static boolean canForm(
            String word,
            Map<Character, Integer> source) {

        Map<Character, Integer> current =
                new HashMap<>(source);

        for (char ch : word.toCharArray()) {

            if (current.getOrDefault(ch, 0) == 0) {
                return false;
            }

            current.put(
                    ch,
                    current.get(ch) - 1
            );
        }

        return true;
    }
}
```

## 64. Optimized Solution Using Array Frequency

```java
public class CountCharactersOptimized {

    public static void main(String[] args) {

        String[] words =
                {"cat", "bt", "hat", "tree"};

        String chars = "atach";

        System.out.println(
                countCharacters(words, chars)
        );
    }

    static int countCharacters(
            String[] words,
            String chars) {

        int[] freq = new int[26];

        for (char ch : chars.toCharArray()) {
            freq[ch - 'a']++;
        }

        int result = 0;

        for (String word : words) {

            int[] temp = freq.clone();

            boolean valid = true;

            for (char ch : word.toCharArray()) {

                if (--temp[ch - 'a'] < 0) {
                    valid = false;
                    break;
                }
            }

            if (valid) {
                result += word.length();
            }
        }

        return result;
    }
}
```

## 65. Interview Follow-Up: Why Two Pointers (i and j)?

```java
// i -> scans every element
// j -> points to next position for non-zero

int[] nums = {0, 1, 0, 3, 12};

j = 0

i = 0 -> nums[0] = 0 -> skip

i = 1 -> nums[1] = 1
swap(i, j)
[1, 0, 0, 3, 12]
j++

i = 2 -> nums[2] = 0 -> skip

i = 3 -> nums[3] = 3
swap(i, j)
[1, 3, 0, 0, 12]
j++

i = 4 -> nums[4] = 12
swap(i, j)
[1, 3, 12, 0, 0]
j++
```
