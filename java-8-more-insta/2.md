## 51. Find K Largest Elements Using Priority Queue (Min Heap)

```java
import java.util.PriorityQueue;
import java.util.Arrays;

public class KLargestElements {

    public static void main(String[] args) {

        int[] nums = {10, 4, 8, 20, 15, 7};

        int k = 3;

        int[] result = getKLargest(nums, k);

        System.out.println(Arrays.toString(result));
    }

    static int[] getKLargest(
            int[] nums,
            int k) {

        // Min heap maintains top K elements
        PriorityQueue<Integer> minHeap =
                new PriorityQueue<>();

        for (int num : nums) {

            minHeap.offer(num);

            // Remove smallest if size exceeds K
            if (minHeap.size() > k) {
                minHeap.poll();
            }
        }

        return minHeap.stream()
                .mapToInt(Integer::intValue)
                .toArray();
    }
}
```

## 52. Compare Version Numbers

```java
public class VersionComparison {

    public static void main(String[] args) {

        String v1 = "1.2";
        String v2 = "1.10";

        System.out.println(
                compareVersion(v1, v2)
        );
    }

    static int compareVersion(
            String v1,
            String v2) {

        String[] a = v1.split("\\.");
        String[] b = v2.split("\\.");

        int length =
                Math.max(a.length, b.length);

        for (int i = 0; i < length; i++) {

            int n1 =
                    i < a.length
                            ? Integer.parseInt(a[i])
                            : 0;

            int n2 =
                    i < b.length
                            ? Integer.parseInt(b[i])
                            : 0;

            if (n1 < n2) {
                return -1;
            }

            if (n1 > n2) {
                return 1;
            }
        }

        return 0;
    }
}
```

## 53. Check Trionic Array

```java
public class TrionicArray {

    public static void main(String[] args) {

        int[] nums =
                {1, 3, 5, 4, 2, 6, 8};

        System.out.println(
                isTrionic(nums)
        );
    }

    static boolean isTrionic(
            int[] nums) {

        int n = nums.length;

        if (n < 4) {
            return false;
        }

        int i = 0;

        // Increasing
        while (i < n - 1 &&
                nums[i] < nums[i + 1]) {
            i++;
        }

        if (i == 0) {
            return false;
        }

        int peak = i;

        // Decreasing
        while (i < n - 1 &&
                nums[i] > nums[i + 1]) {
            i++;
        }

        if (i == peak) {
            return false;
        }

        int valley = i;

        // Increasing again
        while (i < n - 1 &&
                nums[i] < nums[i + 1]) {
            i++;
        }

        return i == n - 1
                && valley < n - 1;
    }
}
```

## 56. Find Missing Number Using XOR

```java
public class MissingNumberXor {

    public static void main(String[] args) {

        int[] arr = {0, 1, 2, 4, 5};

        int n = arr.length;

        int xor = 0;

        // XOR all indexes
        for (int i = 0; i <= n; i++) {
            xor ^= i;
        }

        // XOR all array values
        for (int num : arr) {
            xor ^= num;
        }

        System.out.println(xor);
    }
}
```

## 59. Find Top K Largest Elements (Sorted Output)

```java
import java.util.PriorityQueue;
import java.util.Arrays;

public class TopKLargest {

    public static void main(String[] args) {

        int[] nums =
                {10, 4, 15, 7, 20, 30};

        int k = 3;

        PriorityQueue<Integer> minHeap =
                new PriorityQueue<>();

        for (int num : nums) {

            minHeap.offer(num);

            if (minHeap.size() > k) {
                minHeap.poll();
            }
        }

        int[] result =
                minHeap.stream()
                        .sorted()
                        .mapToInt(Integer::intValue)
                        .toArray();

        System.out.println(
                Arrays.toString(result)
        );
    }
}
```

## 60. Find Kth Largest Element

```java
import java.util.PriorityQueue;

public class KthLargestElement {

    public static void main(String[] args) {

        int[] nums =
                {10, 4, 15, 7, 20, 30};

        int k = 3;

        PriorityQueue<Integer> minHeap =
                new PriorityQueue<>();

        for (int num : nums) {

            minHeap.offer(num);

            if (minHeap.size() > k) {
                minHeap.poll();
            }
        }

        // Heap root contains kth largest
        System.out.println(
                minHeap.peek()
        );
    }
}
```
